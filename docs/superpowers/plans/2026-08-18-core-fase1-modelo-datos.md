```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Rol {
  cliente
  medico
  backoffice
}

enum TipoOrganizacion {
  isapre
  compin
  empresa
  seguro
  subcontratista
}

enum EstadoCaso {
  recibido
  en_revision
  informe_en_validacion
  entregado
}

enum PrioridadCaso {
  normal
  urgente
}

enum EstadoSesion {
  agendada
  en_curso
  realizada
  incumplida_medico
  incumplida_evaluado
  reprogramada
  cancelada
}

enum FirmaProveedor {
  firmaweb
  sovos
  otro
}

model Usuario {
  id             String   @id @default(cuid())
  nombre         String
  email          String   @unique
  passwordHash   String
  rol            Rol
  activo         Boolean  @default(true)
  creadoEn       DateTime @default(now())
  organizacionId String?
  especialidad   String?

  organizacion      Organizacion? @relation(fields: [organizacionId], references: [id], onDelete: Restrict)
  casosAsignados    Caso[]        @relation("CasoMedico")
  informesGenerados Informe[]     @relation("InformeGeneradoPor")
  descargas         LogDescarga[]

  @@index([email])
  @@index([organizacionId])
}

model Organizacion {
  id           String           @id @default(cuid())
  nombre       String
  tipo         TipoOrganizacion
  plazoSlaDias Int
  creadoEn     DateTime         @default(now())

  usuarios Usuario[]
  casos    Caso[]
}

model Caso {
  id             String        @id @default(cuid())
  organizacionId String
  medicoId       String?
  estado         EstadoCaso    @default(recibido)
  tipoLicencia   String
  fechaIngreso   DateTime
  fechaLimite    DateTime
  prioridad      PrioridadCaso @default(normal)
  creadoEn       DateTime      @default(now())
  actualizadoEn  DateTime      @updatedAt

  organizacion Organizacion @relation(fields: [organizacionId], references: [id], onDelete: Restrict)
  medico       Usuario?     @relation("CasoMedico", fields: [medicoId], references: [id], onDelete: Restrict)
  sesion       Sesion?
  informe      Informe?

  @@index([fechaLimite])
  @@index([organizacionId])
  @@index([estado])
}

model Sesion {
  id                       String       @id @default(cuid())
  casoId                   String       @unique
  fechaProgramada          DateTime
  dailyRoomUrl             String?
  estado                   EstadoSesion @default(agendada)
  consentimientoTimestamp  DateTime?
  grabacionUrl             String?
  medicoHoraConexion       DateTime?
  medicoHoraDesconexion    DateTime?
  evaluadoHoraConexion     DateTime?
  evaluadoHoraDesconexion  DateTime?
  duracionEfectivaSegundos Int?

  caso Caso @relation(fields: [casoId], references: [id], onDelete: Restrict)
}

model Informe {
  id                String         @id @default(cuid())
  casoId            String         @unique
  archivoUrl        String
  generadoEn        DateTime       @default(now())
  generadoPor       String
  firmaProveedor    FirmaProveedor
  firmaTimestamp    DateTime?
  firmaDocumentoId  String?
  archivoFirmadoUrl String?

  caso      Caso          @relation(fields: [casoId], references: [id], onDelete: Restrict)
  medico    Usuario       @relation("InformeGeneradoPor", fields: [generadoPor], references: [id], onDelete: Restrict)
  descargas LogDescarga[]
}

model LogDescarga {
  id        String   @id @default(cuid())
  informeId String
  usuarioId String
  timestamp DateTime @default(now())

  informe Informe @relation(fields: [informeId], references: [id], onDelete: Restrict)
  usuario Usuario @relation(fields: [usuarioId], references: [id], onDelete: Restrict)
}
```

Note: `Organizacion` rows use fixed, deterministic `id` values (`seed-org-isapre`, `seed-org-empresa`) instead of the default `cuid()` — `Organizacion` has no other unique field to `upsert` on, and a fixed id keeps the seed idempotent across repeated runs, matching the existing `Usuario` upsert-by-email pattern.

No routes or business logic were added — Fase 2 (`/cliente/*` read-only portal, `/admin/casos/nuevo` + `/admin/casos/importar`, automatic assignment by lowest active caseload) builds directly on this schema. Per the master prompt's working rules, **stop here and report back** before starting Fase 2.

Note: unlike Fase 0, this phase has no VPS/deploy task — the schema change reaches production the same way any other code change does, via the existing GitHub Actions pipeline (`npx prisma migrate deploy` already runs on every deploy, per Fase 0's `.github/workflows/deploy.yml`). No manual production steps are needed once this branch merges to `main`.

**Seed data does NOT reach production.** `npx prisma migrate deploy` (what CI runs) never executes `prisma/seed.ts` — only `migrate dev` and `prisma db seed` do, and neither runs in the deploy workflow. The idempotent seed data (2 `Organizacion`, 2 `medico`, 2 `cliente`, plus Fase 0's `backoffice` user) exists only in local dev. `prisma/seed.ts` now refuses to run when `NODE_ENV=production` (see Task 1/2 fix history) specifically to prevent the Fase 0 precedent — SSHing into the VPS and running `npm run seed` directly — from ever pushing these shared-password (`ChangeMe123!`) test accounts, including two `cliente`-role logins, into a system handling health data under Ley 21.719. **Whoever picks up Fase 2 must create a real `cliente` test/demo user in production deliberately** (not via this seed script) if one is needed to smoke-test the `/cliente/*` portal.
